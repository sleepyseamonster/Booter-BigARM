using System.IO;
using System.Threading;
using BooterBigArm.TopDown3D;
using BooterBigArm.TopDown3D.WorldCreator;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering.Universal;

namespace BooterBigArm.Editor.WorldCreator
{
    public static class TempCanyonRender
    {
        public static void Run()
        {
            var tier = System.Environment.GetEnvironmentVariable("CANYON_RENDER_TIER") ?? "near";
            var actualMid = tier == "actual-mid";
            var actualFar = tier == "actual-far";
            var pose = System.Environment.GetEnvironmentVariable("CANYON_RENDER_POSE") ?? "overview";
            var decorate = System.Environment.GetEnvironmentVariable("CANYON_RENDER_DECORATE") != "0";
            var whiteGroup = System.Environment.GetEnvironmentVariable("CANYON_RENDER_WHITE_GROUP") ?? "";
            var contactProbe = System.Environment.GetEnvironmentVariable("CANYON_RENDER_CONTACT_PROBE") == "1";
            var coarseQuads = tier == "far" ? 1 : tier == "mid" ? 4 : 12;
            var settings = AssetDatabase.LoadAssetAtPath<TopDown3DWorldSettings>(
                "Assets/_Project/Settings/World/TopDown3DWorldSettings.asset");
            using var runtime = WorldCreatorProductionRuntime.Create(24681357L);
            var generator = new TopDown3DWorldGenerator(settings, runtime);
            var representationProfile =
                System.Environment.GetEnvironmentVariable("CANYON_RENDER_DENSE_TIERS") == "1"
                    ? new WorldRepresentationBuildProfile(49, 33, 33)
                    : runtime.Profile.CreateRepresentationProfile();
            var compiler = new WorldRepresentationCompiler(runtime.Query,
                runtime.Authority.Materials, new WorldRepresentationBufferPool(),
                representationProfile, runtime.SourceFingerprint);
            if (System.Environment.GetEnvironmentVariable("CANYON_RENDER_SAMPLE_GRID") == "1")
            {
                var probe = new AbsoluteWorldPosition(488.74d, 0d, 342.39d);
                if (runtime.Query.TrySampleSurface(probe, out var probeSurface, out var probeError)
                    && runtime.Query.TrySampleAffordance(probe, WorldAgentProfile.BooterProof,
                        out var probeAffordance, out probeError))
                    Debug.Log($"CANYON WAYPOINT PROBE semantic={probeSurface.Semantic} height={probeSurface.Position.Vertical:F2} walkable={probeAffordance.Walkable} slope={probeAffordance.SlopeDegrees:F1} reserved={probeAffordance.ReservedRoute}");
                using var writer = new StreamWriter("/tmp/booter-canyon-material-grid.csv");
                writer.WriteLine("x,z,height,semantic,slope,strata,deposit,erosion,packedR,packedG,packedB");
                for (var z = 320; z <= 368; z += 2)
                for (var x = 398; x <= 450; x += 2)
                {
                    var p = new AbsoluteWorldPosition(x, 0d, z);
                    if (!runtime.Query.TrySampleSurface(p, out var s, out var e)
                        || !runtime.Authority.Materials.TrySample(p, out var m, out e))
                        throw new System.InvalidOperationException(e);
                    var packed = WorldTerrainMaterialPackingAdapter.Pack(m);
                    writer.WriteLine($"{x},{z},{s.Position.Vertical:F3},{(int)s.Semantic},{m.SlopeDegrees:F2},{m.StrataExposure:F3},{m.Deposit:F3},{m.Erosion:F3},{packed.r:F3},{packed.g:F3},{packed.b:F3}");
                }
                return;
            }
            var ground = AssetDatabase.LoadAssetAtPath<Material>(
                "Assets/_Project/Materials/TopDown3D/Greybox_Terrain.mat");
            Material materialOverride = null;
            if (System.Environment.GetEnvironmentVariable("CANYON_RENDER_FORCE_FAR_MATERIAL") == "1"
                || System.Environment.GetEnvironmentVariable("CANYON_RENDER_WHITE_ALBEDO") == "1"
                || System.Environment.GetEnvironmentVariable("CANYON_RENDER_WHITE_ROCK") == "1"
                || System.Environment.GetEnvironmentVariable("CANYON_RENDER_DEBUG_GEO") == "1"
                || whiteGroup.Length > 0)
            {
                materialOverride = new Material(ground);
                if (System.Environment.GetEnvironmentVariable("CANYON_RENDER_DEBUG_GEO") == "1")
                    materialOverride.SetFloat("_DebugGeology", 1f);
                if (System.Environment.GetEnvironmentVariable("CANYON_RENDER_FORCE_FAR_MATERIAL") == "1")
                {
                    materialOverride.SetFloat("_DetailFadeStart", 0f);
                    materialOverride.SetFloat("_DetailFadeEnd", 1f);
                }
                if (System.Environment.GetEnvironmentVariable("CANYON_RENDER_WHITE_ALBEDO") == "1")
                    foreach (var property in new[] { "_BaseMap", "_SweptSandMap", "_GravelMap",
                        "_RockyMap", "_RockyTransitionMap", "_RockyMidTransitionMap" })
                        materialOverride.SetTexture(property, Texture2D.whiteTexture);
                if (System.Environment.GetEnvironmentVariable("CANYON_RENDER_WHITE_ROCK") == "1")
                    foreach (var property in new[] { "_RockyMap", "_RockyTransitionMap",
                        "_RockyMidTransitionMap" })
                        materialOverride.SetTexture(property, Texture2D.whiteTexture);
                if (whiteGroup == "sand")
                    foreach (var property in new[] { "_BaseMap", "_SweptSandMap", "_SweptSandTransitionMap" })
                        materialOverride.SetTexture(property, Texture2D.whiteTexture);
                if (whiteGroup == "base")
                    materialOverride.SetTexture("_BaseMap", Texture2D.whiteTexture);
                if (whiteGroup == "swept")
                    materialOverride.SetTexture("_SweptSandMap", Texture2D.whiteTexture);
                if (whiteGroup == "swept-transition")
                    materialOverride.SetTexture("_SweptSandTransitionMap", Texture2D.whiteTexture);
                if (whiteGroup == "gravel")
                    foreach (var property in new[] { "_GravelMap", "_GravelTransitionMap" })
                        materialOverride.SetTexture(property, Texture2D.whiteTexture);
                ground = materialOverride;
            }
            foreach (var point in new[] { new Vector2(417f, 342f),
                new Vector2(416f, 319f), new Vector2(470f, 342f) })
            {
                var sample = generator.Sample(point.x, point.y);
                Debug.Log($"CANYON SURFACE {point} h={sample.Height:F2} sand={sample.SandWeight:F2} gravel={sample.GravelWeight:F2} bedrock={sample.BedrockWeight:F2} deposit={sample.DepositWeight:F2} weather={sample.Weathering:F2} lithology={sample.Lithology:F2} talus={sample.Talus:F2}");
                var absolute = new AbsoluteWorldPosition(point.x, 0d, point.y);
                if (runtime.Authority.Materials.TrySample(absolute,
                    out var material, out var materialError))
                {
                    var packed = WorldTerrainMaterialPackingAdapter.Pack(material);
                    Debug.Log($"CANYON MATERIAL {point} strata={material.StrataExposure:F2} erosion={material.Erosion:F2} deposit={material.Deposit:F2} sediment={material.Sediment:F2} packed={packed}");
                }
                else Debug.LogError(materialError);
            }
            var sceneRoot = new GameObject("Canyon render proof");
            var lightObject = new GameObject("Sun");
            var cameraObject = new GameObject("Camera");
            RenderTexture texture = null;
            Texture2D output = null;
            try
            {
                var minimumA = actualFar ? 0 : actualMid ? 3 : 15;
                var maximumA = actualFar ? 2 : actualMid ? 8 : 31;
                var minimumB = actualFar ? 0 : actualMid ? 2 : 10;
                var maximumB = actualFar ? 2 : actualMid ? 7 : 28;
                var tileSpan = actualFar ? settings.ChunkSize * 16f
                    : actualMid ? settings.ChunkSize * 4f : settings.ChunkSize;
                for (var b = minimumB; b <= maximumB; b++)
                for (var a = minimumA; a <= maximumA; a++)
                {
                    var coordinate = new Vector2Int(a, b);
                    var chunkObject = new GameObject($"Chunk {a},{b}");
                    chunkObject.transform.SetParent(sceneRoot.transform, false);
                    chunkObject.transform.position = new Vector3(
                        a * tileSpan, 0f, b * tileSpan);
                    var detailed = tier == "near"
                        && a >= 21 && a <= 25 && b >= 17 && b <= 20;
                    Mesh mesh;
                    if (actualMid || actualFar || detailed)
                    {
                        var representationTier = actualFar ? WorldRepresentationTier.Far
                            : actualMid ? WorldRepresentationTier.Mid : WorldRepresentationTier.Near;
                        var key = runtime.CreateRepresentationKey(
                            representationTier, a, b, tileSpan);
                        using var result = compiler.BuildAsync(key,
                            CancellationToken.None).GetAwaiter().GetResult();
                        if (actualFar
                            && System.Environment.GetEnvironmentVariable("CANYON_RENDER_ADAPTIVE_FAR") == "1")
                        {
                            mesh = TempCanyonAdaptiveFarMeshBuilder.Build(result,
                                (IWorldCoarseSurfaceQuery)runtime.Query,
                                (IWorldSurfaceMaterialFromSurfaceService)runtime.Authority.Materials,
                                $"Adaptive far {a},{b}", out var adaptiveMetrics, out _);
                            Debug.Log($"CANYON ADAPTIVE tile={a},{b} candidates={adaptiveMetrics.CandidateCells} selected={adaptiveMetrics.SelectedCells} vertices={adaptiveMetrics.Vertices} triangles={adaptiveMetrics.Triangles} extraSamples={adaptiveMetrics.ExtraSurfaceSamples} elapsedMs={adaptiveMetrics.ElapsedMs:F2}");
                        }
                        else mesh = TopDown3DChunkMeshBuilder.BuildMesh(result,
                            $"Actual representation {representationTier} {a},{b}");
                    }
                    else mesh = BuildCoarseMesh(settings, runtime, coordinate, coarseQuads);
                    chunkObject.AddComponent<MeshFilter>().sharedMesh = mesh;
                    chunkObject.AddComponent<MeshRenderer>().sharedMaterial = ground;
                    if (contactProbe)
                        chunkObject.AddComponent<MeshCollider>().sharedMesh = mesh;
                    if (!actualMid && !actualFar)
                    {
                        var chunk = chunkObject.AddComponent<TopDown3DGeneratedChunk>();
                        chunk.Initialize(coordinate, mesh);
                        if (detailed && decorate)
                            foreach (var _ in TopDown3DCliffFaceDecorator.DecorateSteps(
                                chunk, settings, runtime, Vector2.zero, null)) { }
                    }
                }

                var light = lightObject.AddComponent<Light>();
                light.type = LightType.Directional;
                if (System.Environment.GetEnvironmentVariable("CANYON_RENDER_SCENE_LIGHT") == "1")
                {
                    light.intensity = 1.1011368f;
                    light.color = new Color(1f, 0.7253322f, 0.46321848f);
                    lightObject.transform.rotation = new Quaternion(
                        0.20241739f, -0.18498775f, 0.03896926f, 0.96087885f);
                    RenderSettings.ambientLight =
                        System.Environment.GetEnvironmentVariable("CANYON_RENDER_NEUTRAL_AMBIENT") == "1"
                            ? new Color(0.46f, 0.44f, 0.42f)
                            : new Color(0.38f, 0.23f, 0.17f);
                }
                else
                {
                    light.intensity = 1.7f;
                    light.color = new Color(1f, 0.78f, 0.65f);
                    lightObject.transform.rotation = Quaternion.Euler(35f, -35f, 0f);
                    RenderSettings.ambientLight = new Color(0.52f, 0.38f, 0.33f);
                }
                var camera = cameraObject.AddComponent<Camera>();
                var cameraData = cameraObject.AddComponent<UniversalAdditionalCameraData>();
                cameraData.SetRenderer(1);
                cameraData.renderPostProcessing = true;
                if (pose == "game-floor" || pose == "game-rim")
                {
                    var point = pose == "game-floor"
                        ? new Vector2(417f, 342f) : new Vector2(416f, 319f);
                    var target = new Vector3(point.x,
                        generator.Sample(point.x, point.y).Height + 1.15f, point.y);
                    var rotation = Quaternion.Euler(50f, 40f, 0f);
                    camera.transform.SetPositionAndRotation(
                        target - rotation * Vector3.forward * 25f, rotation);
                    camera.fieldOfView = 48f;
                }
                else
                {
                    camera.transform.position = new Vector3(535f, 165f, 212f);
                    camera.transform.LookAt(new Vector3(417f, -9f, 343f));
                    camera.fieldOfView = 58f;
                }
                camera.nearClipPlane = 0.3f;
                camera.farClipPlane = 500f;
                camera.aspect = 1600f / 900f;
                camera.clearFlags = CameraClearFlags.SolidColor;
                camera.backgroundColor = new Color(0.74f, 0.48f, 0.37f);
                if (contactProbe)
                {
                    Physics.SyncTransforms();
                    foreach (var pixel in new[] { new Vector2(500f, 600f),
                        new Vector2(700f, 450f), new Vector2(900f, 300f),
                        new Vector2(600f, 300f), new Vector2(1100f, 500f) })
                    {
                        var ray = camera.ViewportPointToRay(new Vector3(
                            pixel.x / 1600f, 1f - pixel.y / 900f, 0f));
                        var hits = Physics.RaycastAll(ray, 500f,
                            ~0, QueryTriggerInteraction.Ignore);
                        System.Array.Sort(hits, (a, b) => a.distance.CompareTo(b.distance));
                        for (var hitIndex = 0; hitIndex < Mathf.Min(3, hits.Length); hitIndex++)
                        {
                            var hit = hits[hitIndex];
                            var absolute = new AbsoluteWorldPosition(hit.point.x,
                                0d, hit.point.z);
                            if (!runtime.Query.TrySampleSurface(absolute,
                                    out var surface, out var error))
                                throw new System.InvalidOperationException(error);
                            if (!runtime.Authority.Materials.TrySample(absolute,
                                    out var surfaceMaterial, out error))
                                throw new System.InvalidOperationException(error);
                            var packed = WorldTerrainMaterialPackingAdapter.Pack(surfaceMaterial);
                            Debug.Log($"CANYON CONTACT pixel={pixel} hit={hit.collider.name} index={hitIndex} world={hit.point} normal={hit.normal} canonicalDelta={hit.point.y - surface.Position.Vertical:F3} distance={hit.distance:F2} semantic={surface.Semantic} slope={surfaceMaterial.SlopeDegrees:F1} strata={surfaceMaterial.StrataExposure:F2} packed={packed}");
                        }
                    }
                }
                if (System.Environment.GetEnvironmentVariable("CANYON_RENDER_RAY_PROBE") == "1")
                {
                    foreach (var pixel in new[] { new Vector2(1000f, 250f), new Vector2(1200f, 260f),
                        new Vector2(1100f, 400f), new Vector2(850f, 390f), new Vector2(1230f, 500f),
                        new Vector2(1250f, 650f), new Vector2(1350f, 500f), new Vector2(1000f, 600f) })
                    {
                        var ray = camera.ViewportPointToRay(new Vector3(pixel.x / 1600f,
                            pixel.y / 900f, 0f));
                        for (var distance = 0f; distance < 500f; distance += 0.25f)
                        {
                            var point = ray.GetPoint(distance);
                            var absolute = new AbsoluteWorldPosition(point.x, 0d, point.z);
                            if (!runtime.Query.TrySampleSurface(absolute, out var surface, out var error))
                                throw new System.InvalidOperationException(error);
                            if (point.y > surface.Position.Vertical) continue;
                            if (!runtime.Authority.Materials.TrySample(absolute, out var material, out error))
                                throw new System.InvalidOperationException(error);
                            var packed = WorldTerrainMaterialPackingAdapter.Pack(material);
                            Debug.Log($"CANYON RAY pixel={pixel} world=({point.x:F2},{point.y:F2},{point.z:F2}) semantic={surface.Semantic} slope={material.SlopeDegrees:F1} strata={material.StrataExposure:F2} deposit={material.Deposit:F2} packed={packed}");
                            break;
                        }
                    }
                }
                texture = new RenderTexture(1600, 900, 24);
                camera.targetTexture = texture;
                camera.Render();
                RenderTexture.active = texture;
                output = new Texture2D(1600, 900, TextureFormat.RGB24, false);
                output.ReadPixels(new Rect(0, 0, 1600, 900), 0, 0);
                output.Apply();
                var outputPath = "/tmp/booter-canyon-" + tier + "-" + pose
                    + (whiteGroup.Length > 0 ? "-white-" + whiteGroup : "")
                    + (System.Environment.GetEnvironmentVariable("CANYON_RENDER_DEBUG_GEO") == "1"
                        ? "-debug-geology" : "") + ".png";
                File.WriteAllBytes(outputPath, output.EncodeToPNG());
                Debug.Log("CANYON RENDER " + outputPath);
            }
            finally
            {
                RenderTexture.active = null;
                Object.DestroyImmediate(output);
                Object.DestroyImmediate(texture);
                Object.DestroyImmediate(materialOverride);
                Object.DestroyImmediate(cameraObject);
                Object.DestroyImmediate(lightObject);
                Object.DestroyImmediate(sceneRoot);
            }
        }

        private static Mesh BuildCoarseMesh(TopDown3DWorldSettings settings,
            WorldCreatorProductionRuntime runtime, Vector2Int coordinate, int quads)
        {
            var vertices = new Vector3[(quads + 1) * (quads + 1)];
            var colors = new Color[vertices.Length];
            var triangles = new int[quads * quads * 6];
            var step = settings.ChunkSize / quads;
            for (var z = 0; z <= quads; z++)
            for (var x = 0; x <= quads; x++)
            {
                var localX = x * step;
                var localZ = z * step;
                var absolute = new AbsoluteWorldPosition(
                    coordinate.x * settings.ChunkSize + localX,
                    0d, coordinate.y * settings.ChunkSize + localZ);
                if (!runtime.Query.TrySampleSurface(absolute, out var surface, out var error))
                    throw new System.InvalidOperationException(error);
                if (!runtime.Authority.Materials.TrySample(absolute,
                    out var material, out error))
                    throw new System.InvalidOperationException(error);
                vertices[z * (quads + 1) + x] = new Vector3(localX,
                    (float)surface.Position.Vertical, localZ);
                colors[z * (quads + 1) + x] =
                    WorldTerrainMaterialPackingAdapter.Pack(material);
            }
            var index = 0;
            for (var z = 0; z < quads; z++)
            for (var x = 0; x < quads; x++)
            {
                var bottom = z * (quads + 1) + x;
                var top = bottom + quads + 1;
                triangles[index++] = bottom;
                triangles[index++] = top;
                triangles[index++] = bottom + 1;
                triangles[index++] = bottom + 1;
                triangles[index++] = top;
                triangles[index++] = top + 1;
            }
            var mesh = new Mesh();
            mesh.vertices = vertices;
            mesh.triangles = triangles;
            mesh.colors = colors;
            mesh.RecalculateNormals();
            mesh.RecalculateBounds();
            return mesh;
        }
    }
}
