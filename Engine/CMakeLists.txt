cmake_minimum_required(VERSION 3.20)
project(BooterBigARMEngine VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT APPLE AND NOT WIN32)
    message(FATAL_ERROR "This foundation currently supports macOS and Windows only")
endif()
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Tools/prepare_probe.py"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" RESULT_VARIABLE source_result)
if(NOT source_result EQUAL 0)
    message(FATAL_ERROR "Pinned sources unavailable or changed. Run Tools/prepare_probe.py --fetch and inspect its result.")
endif()
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/.cache/probe-sources")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/Research/runtime-lock.json" RUNTIME_LOCK_HASH)
add_library(engine_core Source/Core/WorldIdentity.cpp Source/Persistence/Document.cpp)
target_include_directories(engine_core PUBLIC Source)
target_include_directories(engine_core SYSTEM PUBLIC "${DEPS}/nlohmann_json/single_include")
add_library(engine_state Source/Core/FixtureState.cpp Source/Core/FixtureGeometry.cpp Source/Runtime/InspectionDocument.cpp)
target_include_directories(engine_state PUBLIC Source)
target_link_libraries(engine_state PUBLIC engine_core)
option(ENGINE_BUILD_WORKBENCH "Build SDL/bgfx workbench and shaders" ON)
if(ENGINE_BUILD_WORKBENCH)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/sdl" "${CMAKE_BINARY_DIR}/sdl" EXCLUDE_FROM_ALL)
set(BGFX_DIR "${DEPS}/bgfx" CACHE STRING "" FORCE)
set(BX_DIR "${DEPS}/bx" CACHE STRING "" FORCE)
set(BIMG_DIR "${DEPS}/bimg" CACHE STRING "" FORCE)
set(BGFX_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_SHADER ON CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_BIN2C OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_GEOMETRY OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_TEXTURE OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BGFX_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/bgfx_cmake" "${CMAKE_BINARY_DIR}/bgfx" EXCLUDE_FROM_ALL)
set(IMGUI_DIR "${DEPS}/imgui")
add_library(engine_imgui STATIC "${IMGUI_DIR}/imgui.cpp" "${IMGUI_DIR}/imgui_draw.cpp"
    "${IMGUI_DIR}/imgui_tables.cpp" "${IMGUI_DIR}/imgui_widgets.cpp"
    "${IMGUI_DIR}/backends/imgui_impl_sdl3.cpp")
target_include_directories(engine_imgui PUBLIC "${IMGUI_DIR}" "${IMGUI_DIR}/backends")
target_link_libraries(engine_imgui PUBLIC SDL3::SDL3-static)
include(CMake/Shaders.cmake)
add_executable(engine_workbench Apps/Workbench/main.cpp Source/Platform/Window.cpp
    Source/Rendering/Renderer.cpp Source/Rendering/InspectorRenderer.cpp)
target_link_libraries(engine_workbench PRIVATE engine_state engine_imgui bgfx bx)
target_include_directories(engine_workbench PRIVATE Source)
target_compile_definitions(engine_workbench PRIVATE ENGINE_BUILD_ID="${PROJECT_VERSION}-${RUNTIME_LOCK_HASH}")
add_dependencies(engine_workbench engine_shaders)
if(MSVC)
    target_compile_options(engine_state PRIVATE /W4)
    target_compile_options(engine_workbench PRIVATE /W4)
else()
    target_compile_options(engine_state PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(engine_workbench PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET engine_workbench POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_BINARY_DIR}/Shaders" "$<TARGET_FILE_DIR:engine_workbench>/Shaders"
    VERBATIM)
install(TARGETS engine_workbench RUNTIME DESTINATION bin)
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Shaders/" DESTINATION bin/Shaders)
install(DIRECTORY Research/Licenses/ DESTINATION share/engine/licenses)
endif()
enable_testing()
add_executable(engine_state_tests Tests/FixtureStateTests.cpp)
target_link_libraries(engine_state_tests PRIVATE engine_state)
add_test(NAME foundation_state COMMAND engine_state_tests)
add_executable(engine_geometry_tests Tests/FixtureGeometryTests.cpp)
target_link_libraries(engine_geometry_tests PRIVATE engine_state)
add_test(NAME fixture_geometry COMMAND engine_geometry_tests)

add_executable(engine_core_tests Tests/CoreTests.cpp)
target_link_libraries(engine_core_tests PRIVATE engine_state)
add_test(NAME runtime_documents_identity COMMAND engine_core_tests "${CMAKE_CURRENT_BINARY_DIR}/core-test-data")
