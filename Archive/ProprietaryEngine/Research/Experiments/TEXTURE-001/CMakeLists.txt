cmake_minimum_required(VERSION 3.20)
project(texture_research LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PROBE_SOURCES "${ENGINE_ROOT}/.cache/probe-sources")
set(BGFX_DIR "${PROBE_SOURCES}/bgfx" CACHE PATH "" FORCE)
set(BX_DIR "${PROBE_SOURCES}/bx" CACHE PATH "" FORCE)
set(BIMG_DIR "${PROBE_SOURCES}/bimg" CACHE PATH "" FORCE)
set(BGFX_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_TEXTURE ON CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_SHADER OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_GEOMETRY OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_BIN2C OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BGFX_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${PROBE_SOURCES}/bgfx_cmake" dependencies EXCLUDE_FROM_ALL)
add_executable(texture_inspect inspect.cpp)
target_link_libraries(texture_inspect PRIVATE bimg)
